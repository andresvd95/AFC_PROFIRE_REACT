document.addEventListener('DOMContentLoaded', function () {
      const form = document.getElementById('contactFormEmailJS');
      const btn = document.getElementById('contactSubmitBtn');
      const result = document.getElementById('contactResult');

      // Campos
      const elName = document.getElementById('contact-name');
      const elEmail = document.getElementById('contact-email');
      const elPhone = document.getElementById('contact-phone');
      const elPref = document.getElementById('contact-pref');
      const elMessage = document.getElementById('contact-message');

      // EmailJS
      const SERVICE_ID = 'service_433wpss';
      const TEMPLATE_ID = 'template_0rq3a9q';

      // Helpers de validación
      const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/i;
      const PHONE_RE = /^[0-9+\-\s()]{7,20}$/; // acepta +, -, espacios, (), 7-20 chars

      function clearErrors() {
        result.innerHTML = '';
        [elName, elEmail, elPhone, elPref, elMessage].forEach(el => {
          el.classList.remove('is-invalid');
          el.removeAttribute('aria-invalid');
          el.setCustomValidity('');
        });
      }

      function markInvalid(el, msg) {
        el.classList.add('is-invalid');
        el.setAttribute('aria-invalid', 'true');
        el.setCustomValidity(msg);
      }

      form.addEventListener('submit', function (e) {
        e.preventDefault();
        clearErrors();

        // Valores
        const name = elName.value.trim();
        const email = elEmail.value.trim();
        const phone = elPhone.value.trim();
        const pref = elPref.value;
        const msg = elMessage.value.trim();

        // Validaciones
        let firstInvalid = null;
        let messages = [];

        if (!name || name.length < 2) {
          markInvalid(elName, 'Ingresa tu nombre.');
          messages.push('• Nombre es obligatorio.');
          firstInvalid = firstInvalid || elName;
        }

        if (!email || !EMAIL_RE.test(email)) {
          markInvalid(elEmail, 'Correo inválido.');
          messages.push('• Ingresa un correo válido.');
          firstInvalid = firstInvalid || elEmail;
        }

        if (!phone || !PHONE_RE.test(phone)) {
          markInvalid(elPhone, 'Teléfono inválido (mín. 7 dígitos).');
          messages.push('• Ingresa un teléfono válido (7 a 20 caracteres, puede incluir + - () y espacios).');
          firstInvalid = firstInvalid || elPhone;
        }

        if (!pref) {
          markInvalid(elPref, 'Selecciona un medio de contacto.');
          messages.push('• Selecciona el medio de contacto.');
          firstInvalid = firstInvalid || elPref;
        }

        if (!msg || msg.length < 10) {
          markInvalid(elMessage, 'Escribe un mensaje (mín. 10 caracteres).');
          messages.push('• El mensaje debe tener al menos 10 caracteres.');
          firstInvalid = firstInvalid || elMessage;
        }

        // Si hay errores, mostramos y detenemos
        if (messages.length) {
          result.innerHTML = '<div class="alert alert-danger" role="alert">' +
            'Por favor corrige:<br>' + messages.join('<br>') +
            '</div>';
          // Muestra tooltip nativo si existe
          if (typeof form.reportValidity === 'function') {
            form.reportValidity();
          }
          // Enfoca el primer campo inválido
          if (firstInvalid && typeof firstInvalid.focus === 'function') {
            firstInvalid.focus();
          }
          return;
        }

        // Sin errores -> enviar
        const originalBtnHTML = btn.innerHTML;
        btn.disabled = true;
        btn.innerHTML = '<span>Enviando…</span>';

        const templateParams = {
          user_name: name,
          user_email: email,
          user_phone: phone,
          contact_pref: pref,
          user_message: msg
        };

        emailjs.send(SERVICE_ID, TEMPLATE_ID, templateParams)
          .then(() => {
            result.innerHTML = '<div class="alert alert-success" role="alert">¡Gracias! Tu mensaje fue enviado correctamente.</div>';
            form.reset();
          })
          .catch((err) => {
            console.error('EmailJS error:', err);
            result.innerHTML = '<div class="alert alert-danger" role="alert">No se pudo enviar el mensaje. Intenta más tarde o escríbenos por WhatsApp.</div>';
          })
          .finally(() => {
            btn.disabled = false;
            btn.innerHTML = originalBtnHTML;
            clearErrors(); // limpia estilos de error por si quedaron
          });
      });
    });
